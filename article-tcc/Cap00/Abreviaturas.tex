\chapter*{Lista de Abreviações}

\begin{description}
    \item \textbf{AC} \quad \textit{Accepted} -- Submissão Aceita (aprovada em todos os casos de teste)
    \item \textbf{API} \quad \textit{Application Programming Interface} -- Interface de Programação de Aplicações
    \item \textbf{CE} \quad \textit{Compilation Error} -- Erro de Compilação (ou sintaxe)
    \item \textbf{CoT} \quad \textit{Chain-of-Thought} -- Cadeia de Pensamento
    \item \textbf{CPU} \quad \textit{Central Processing Unit} -- Unidade Central de Processamento
    \item \textbf{CSV} \quad \textit{Comma-Separated Values} -- Valores Separados por Vírgula
    \item \textbf{E/S} \quad Entrada e Saída (\textit{Input/Output})
    \item \textbf{ENADE} \quad Exame Nacional de Desempenho dos Estudantes
    \item \textbf{ENEM} \quad Exame Nacional do Ensino Médio
    \item \textbf{FS} \quad \textit{Few-Shot} -- Aprendizado com Poucos Exemplos
    \item \textbf{JSON} \quad \textit{JavaScript Object Notation} -- Notação de Objetos JavaScript
    \item \textbf{LLM} \quad \textit{Large Language Model} -- Modelo de Linguagem de Grande Porte
    \item \textbf{LOC} \quad \textit{Lines of Code} -- Linhas de Código
    \item \textbf{MLE} \quad \textit{Memory Limit Exceeded} -- Limite de Memória Excedido
    \item \textbf{MLLM} \quad \textit{Multimodal Large Language Model} -- Modelo Multimodal de Linguagem de Grande Porte
    \item \textbf{OBI} \quad Olimpíada Brasileira de Informática
    \item \textbf{PDF} \quad \textit{Portable Document Format} -- Formato Portátil de Documento
    \item \textbf{PICOC} \quad \textit{Population, Intervention, Comparison, Outcome, Context} -- População, Intervenção, Comparação, Resultado e Contexto
    \item \textbf{POSCOMP} \quad Exame Nacional para Ingresso na Pós-Graduação em Computação
    \item \textbf{PP} \quad Pergunta de Pesquisa
    \item \textbf{p.p.} \quad Pontos Percentuais
    \item \textbf{RAM} \quad \textit{Random Access Memory} -- Memória de Acesso Aleatório
    \item \textbf{RE} \quad \textit{Runtime Error} -- Erro em Tempo de Execução
    \item \textbf{SBC} \quad Sociedade Brasileira de Computação
    \item \textbf{STL} \quad \textit{Standard Template Library} -- Biblioteca Padrão de Modelos (C++)
    \item \textbf{TCC} \quad Trabalho de Conclusão de Curso
    \item \textbf{TLE} \quad \textit{Time Limit Exceeded} -- Limite de Tempo Excedido
    \item \textbf{USD} \quad \textit{United States Dollar} -- Dólar Americano
    \item \textbf{VLM} \quad \textit{Vision-Language Model} -- Modelo Multimodal de Visão e Linguagem
    \item \textbf{WA} \quad \textit{Wrong Answer} -- Resposta Incorreta
    \item \textbf{ZS} \quad \textit{Zero-Shot} -- Aprendizado sem Exemplos Prévios
\end{description}
